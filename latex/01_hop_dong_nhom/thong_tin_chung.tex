\section{Thông tin chung và Mục tiêu hoạt động}

\subsection{Thông tin nhóm}
\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{Tên nhóm:} 02
    \item \textbf{Môn học:} Kỹ năng nghề nghiệp (SS004)
    \item \textbf{Mã lớp:} SS004.F31.CN1.CNTT
    \item \textbf{Tên đề tài:} Xây dựng và phát triển trò chơi Tetris (Xếp gạch)
\end{itemize}

\subsection{Mục tiêu hoạt động của nhóm}
\begin{enumerate}
    \item Hoàn thành xuất sắc đồ án môn học Kỹ năng nghề nghiệp đúng thời hạn quy định.
    \item Xây dựng hoàn chỉnh trò chơi Tetris với đầy đủ tính năng và tài liệu kỹ thuật chi tiết.
    \item Áp dụng hiệu quả các kỹ năng mềm (giao tiếp, làm việc nhóm, quản lý thời gian) và kỹ năng chuyên môn (lập trình, quản lý mã nguồn Git, quản lý tiến độ).
    \item Đạt kết quả đánh giá cao từ giảng viên và tối ưu hóa sự phối hợp giữa các thành viên.
\end{enumerate}
