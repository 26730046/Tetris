\section{Các điều khoản cam kết và Quy chế làm việc}

\subsection{Quy tắc làm việc chung}
\begin{itemize}
    \item \textbf{Tôn trọng lẫn nhau:} Lắng nghe ý kiến của các thành viên trong nhóm với tinh thần xây dựng và hòa nhã.
    \item \textbf{Trách nhiệm cá nhân:} Chủ động hoàn thành các nhiệm vụ được phân công đúng hạn và đảm bảo chất lượng.
    \item \textbf{Tính minh bạch:} Cập nhật tiến độ thường xuyên trên Trello/GitHub và thông báo ngay khi gặp khó khăn hoặc sự cố.
    \item \textbf{Tham gia họp nhóm:} Có mặt đúng giờ trong tất cả các buổi họp nhóm (online/offline). Trường hợp vắng mặt phải báo trước ít nhất 12 giờ.
\end{itemize}

\subsection{Quy chế khen thưởng và kỷ luật}
\begin{itemize}
    \item \textbf{Khen thưởng:} Các thành viên có đóng góp xuất sắc, hỗ trợ đồng đội và hoàn thành trước thời hạn sẽ được cộng điểm đánh giá chéo.
    \item \textbf{Kỷ luật:}
    \begin{itemize}
        \item Trễ hạn (deadline) không lý do chính đáng: Nhắc nhở lần 1, trừ điểm đánh giá từ lần 2.
        \item Vắng mặt không phép trong các buổi họp quan trọng: Trừ điểm chuyên cần của nhóm.
        \item Không hoàn thành nhiệm vụ dẫn đến ảnh hưởng tiến độ chung: Xem xét giảm tỷ lệ phần trăm đóng góp trong báo cáo cuối kỳ.
    \end{itemize}
\end{itemize}
