\begin{tikzpicture}[

    startstop/.style={rectangle, rounded corners=8pt, draw=red!60!black, fill=red!8, thick,

    process/.style={rectangle, draw=blue!60!black, fill=blue!6, thick,
                    minimum height=0.75cm, align=center, text width=4cm},
    decision/.style={diamond, aspect=2.2, draw=orange!70!black, fill=orange!10, thick,

    arr/.style={-{Stealth[length=2.3mm]}, thick},
    lbl/.style={font=\footnotesize}
]
    \node[startstop] (s)     at (0, 0)      {Khối vừa được cố định};
    \node[process]   (init)  at (0, -1.3)   {Số dòng xóa $\gets 0$;\\ $i \gets H-2$ (hàng dưới cùng)};
    \coordinate      (join)  at (0, -2.25);
    \node[decision]  (cond)  at (0, -3.2)   {$i \geq 1$?};
    \node[decision]  (full)  at (0, -5.0)   {Hàng $i$\\ đầy?};
    \node[process, text width=2.4cm] (dec) at (0, -6.7) {$i \gets i - 1$};
    \node[process, text width=4.8cm] (shift) at (5.6, -5.0)
        {Tăng số dòng xóa; dồn các hàng\\ từ $i-1$ đến 1 xuống một bậc;\\ làm trống hàng 1;\\ vẽ lại, dừng 100\,ms};
    \node[process, text width=3.6cm] (score) at (-5.0, -5.0)
        {Cộng điểm theo số dòng;\\ cập nhật tổng dòng,\\ cấp độ và tốc độ rơi};
    \node[startstop] (e) at (-5.0, -6.7) {Kết thúc};

    \draw[arr] (s) -- (init);
    \draw[arr] (init) -- (cond);
    \draw[arr] (cond) -- node[lbl, right]{Có} (full);
    \draw[arr] (cond.west) -| node[lbl, above, pos=0.25]{Không} (score.north);
    \draw[arr] (full) -- node[lbl, above]{Có} (shift);
    \draw[arr] (full) -- node[lbl, right]{Không} (dec);
    \draw[arr] (shift.north) |- node[lbl, above, pos=0.7]{giữ nguyên $i$, xét lại} (join);
    \draw[arr] (dec.east) -- (8.6, -6.7) |- (join);
    \draw[arr] (score) -- (e);
