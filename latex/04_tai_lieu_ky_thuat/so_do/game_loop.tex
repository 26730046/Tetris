\documentclass[border=6pt]{standalone}
\usepackage[utf8]{inputenc}
\usepackage[T5]{fontenc}
\usepackage{tikz}
\usetikzlibrary{shapes.geometric, arrows.meta, positioning}

\begin{document}
\begin{tikzpicture}[
    font=\small,
    startstop/.style={rectangle, rounded corners=8pt, draw=red!60!black, fill=red!8, thick,
                      minimum width=2.8cm, minimum height=0.75cm, align=center},
    process/.style={rectangle, draw=blue!60!black, fill=blue!6, thick,
                    minimum height=0.75cm, align=center, text width=4.6cm},
    decision/.style={diamond, aspect=2.4, draw=orange!70!black, fill=orange!10, thick,
                     align=center, inner sep=0pt, text width=2.5cm},
    arr/.style={-{Stealth[length=2.3mm]}, thick},
    lbl/.style={font=\footnotesize}
]
    % Cột chính
    \node[startstop] (start) at (0, 0)      {Bắt đầu khung hình};
    \node[process]   (input) at (0, -1.3)   {Xóa khối đang rơi và bóng mờ khỏi bảng;\\ đọc và xử lý mọi phím đang chờ};
    \node[decision]  (quit)  at (0, -3.0)   {Phím \texttt{q}?};
    \node[process]   (timer) at (0, -4.6)   {Bộ đếm thời gian $+ 30$\,ms};
    \node[decision]  (due)   at (0, -6.1)   {Đủ thời gian\\ rơi một ô?};
    \node[decision]  (fall)  at (0, -8.1)   {Rơi xuống\\ được một ô?};
    \node[process]   (down)  at (0, -9.8)   {Tăng tọa độ $y$ thêm 1};
    \node[process]   (rend)  at (0, -11.3)  {Tính bóng mờ, ghi khối vào bảng,\\ vẽ lại màn hình};
    \node[process]   (sleep) at (0, -12.7)  {Tạm dừng 30\,ms};

    % Cột phải
    \node[startstop] (exit)  at (5.6, -3.0) {Thoát chương trình};
    \node[process, text width=3.8cm] (lock) at (5.6, -8.1) {Cố định khối, xóa dòng đầy,\\ lấy khối kế tiếp từ túi};
    \node[decision, text width=2.2cm] (over) at (5.6, -9.9) {Khối mới\\ đặt được?};
    \node[startstop] (gover) at (10.2, -9.9) {Game Over};

    \draw[arr] (start) -- (input);
    \draw[arr] (input) -- (quit);
    \draw[arr] (quit) -- node[lbl, above]{Có} (exit);
    \draw[arr] (quit) -- node[lbl, right]{Không} (timer);
    \draw[arr] (timer) -- (due);
    \draw[arr] (due) -- node[lbl, right]{Có} (fall);
    \draw[arr] (fall) -- node[lbl, right]{Có} (down);
    \draw[arr] (fall) -- node[lbl, above]{Không} (lock);
    \draw[arr] (lock) -- (over);
    \draw[arr] (over) -- node[lbl, above]{Không} (gover);
    \draw[arr] (over.south) |- node[lbl, pos=0.2, right]{Có} (rend.east);
    \draw[arr] (down) -- (rend);
    \draw[arr] (due.west) -- node[lbl, above]{Không} (-3.4, -6.1) |- (rend.west);
    \draw[arr] (rend) -- (sleep);
    \draw[arr] (sleep.west) -- (-4.2, -12.7) |- (start.west);
\end{tikzpicture}
\end{document}
